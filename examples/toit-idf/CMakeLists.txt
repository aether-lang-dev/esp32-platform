cmake_minimum_required(VERSION 3.16)
set(TOIT_EXAMPLE "" CACHE STRING "Toit example to build")
set(NET_EXAMPLES wifi_connect ntp espnow ethernet ethernet_qemu)
set(BLE_EXAMPLES ble_scan ble_scan_active ble_advertise ble_connect ble_heart_rate)
set(EXAMPLES ${NET_EXAMPLES} ${BLE_EXAMPLES} touch_wake)
if(NOT TOIT_EXAMPLE IN_LIST EXAMPLES)
    message(FATAL_ERROR "Unknown TOIT_EXAMPLE: ${TOIT_EXAMPLE}")
endif()
if(TOIT_EXAMPLE IN_LIST BLE_EXAMPLES)
    set(SDKCONFIG_DEFAULTS "sdkconfig.ble")
elseif(TOIT_EXAMPLE STREQUAL "ethernet")
    set(SDKCONFIG_DEFAULTS "sdkconfig.ethernet")
elseif(TOIT_EXAMPLE STREQUAL "ethernet_qemu")
    set(SDKCONFIG_DEFAULTS "sdkconfig.openeth")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(toit_example)
