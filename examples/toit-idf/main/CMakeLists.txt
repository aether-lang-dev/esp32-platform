if(TOIT_EXAMPLE STREQUAL "wifi_connect" OR TOIT_EXAMPLE STREQUAL "ntp")
    set(src wifi.c)
elseif(TOIT_EXAMPLE STREQUAL "espnow")
    set(src espnow.c)
elseif(TOIT_EXAMPLE MATCHES "^ethernet")
    set(src ethernet.c olimex_poe.c ethernet_provider.c ethernet_client.c)
elseif(TOIT_EXAMPLE STREQUAL "touch_wake")
    set(src touch_wake.c)
elseif(TOIT_EXAMPLE MATCHES "^ble_")
    if(TOIT_EXAMPLE STREQUAL "ble_advertise" OR TOIT_EXAMPLE STREQUAL "ble_heart_rate")
        set(src ble.c ble_peripheral.c)
    else()
        set(src ble.c ble_central.c)
    endif()
endif()
idf_component_register(SRCS ${src} PRIV_REQUIRES esp_wifi esp_event esp_netif esp_timer nvs_flash esp_eth esp_driver_gpio esp_driver_touch_sens bt)
target_compile_definitions(${COMPONENT_LIB} PRIVATE TOIT_${TOIT_EXAMPLE}=1)
target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror)
